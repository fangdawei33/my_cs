空间优化版 (一维数组)
const int N = 1010; int f[N], v[N], w[N];
int n = 4, m = 5;
for (int i = 1; i <= n; ++i) {
for (int j = m; j >= v[i]; j--) {
f[j] = max(f[j], f[j - v[i]] + w[i]);
}
}
