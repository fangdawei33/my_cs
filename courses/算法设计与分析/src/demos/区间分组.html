💻 C++ 核心代码跟踪 O(N log N)
sort(range, range + n); // 按左端点 l 升序
priority_queue<int, vector<int>, greater<int>> heap;
for (int i = 0; i < n; i ++ ) {
auto r = range[i];
if (heap.empty() || heap.top() >= r.l)
heap.push(r.r); // 开新组
else {
heap.pop(); // 弹出最小右端点
heap.push(r.r); // 更新该组右端点
}
}
printf("%d\n", heap.size());
