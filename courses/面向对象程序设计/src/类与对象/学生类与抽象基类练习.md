```cpp
#include <iostream>
#include <string>

using namespace std;

class XS {
public:
    virtual void dispXM() = 0;
    virtual void dispXB() = 0;
    virtual void dispNL() = 0;
};

class CZS : public XS {
public:
    CZS(string, string, int);
    void dispXM();
    void dispXB();
    void dispNL();
private:
    string xm;
    string xb;
    int nl;
};

CZS::CZS(string n, string g, int a) {
    xm = n;
    xb = g;
    nl = a;
}

void CZS::dispXM() {
    cout << xm;
}

void CZS::dispXB() {
    cout << xb;
}

void CZS::dispNL() {
    cout << nl;
}

class GZS : public XS {
public:
    GZS(string, string, int);
    void dispXM();
    void dispXB();
    void dispNL();
private:
    string xm;
    string xb;
    int nl;
};

GZS::GZS(string n, string g, int a) {
    xm = n;
    xb = g;
    nl = a;
}

void GZS::dispXM() {
    cout << xm;
}

void GZS::dispXB() {
    cout << xb;
}

void GZS::dispNL() {
    cout << nl;
}

class DXS : public XS {
public:
    DXS(string, string, int);
    void dispXM();
    void dispXB();
    void dispNL();
private:
    string xm;
    string xb;
    int nl;
};

DXS::DXS(string n, string g, int a) {
    xm = n;
    xb = g;
    nl = a;
}

void DXS::dispXM() {
    cout << xm;
}

void DXS::dispXB() {
    cout << xb;
}

void DXS::dispNL() {
    cout << nl;
}

int main() {
    string n1, g1, n2, g2, n3, g3;
    int a1, a2, a3;

    cin >> n1 >> g1 >> a1;
    cin >> n2 >> g2 >> a2;
    cin >> n3 >> g3 >> a3;

    CZS czs(n1, g1, a1);
    GZS gzs(n2, g2, a2);
    DXS dxs(n3, g3, a3);

    XS *p1 = &czs;
    XS *p2 = &gzs;
    XS *p3 = &dxs;

    p1->dispXM();
    cout << " ";
    p1->dispXB();
    cout << " ";
    p1->dispNL();
    cout << endl;

    p2->dispXM();
    cout << " ";
    p2->dispXB();
    cout << " ";
    p2->dispNL();
    cout << endl;

    p3->dispXM();
    cout << " ";
    p3->dispXB();
    cout << " ";
    p3->dispNL();
    cout << endl;

    XS &r1 = czs;
    XS &r2 = gzs;
    XS &r3 = dxs;

    r1.dispXM();
    cout << " ";
    r1.dispNL();
    cout << " ";
    r1.dispXB();
    cout << endl;

    r2.dispXM();
    cout << " ";
    r2.dispNL();
    cout << " ";
    r2.dispXB();
    cout << endl;

    r3.dispXM();
    cout << " ";
    r3.dispNL();
    cout << " ";
    r3.dispXB();
    cout << endl;

    return 0;
}
```
